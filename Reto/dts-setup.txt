scp fip-stm32mp257f-dk-optee-sdcard.bin root@192.168.7.1:/tmp/

# On target:

SD_DRIVE=$(lsblk -no PKNAME $(findmnt -n -o SOURCE /))
echo "Flashing to SD card at /dev/${SD_DRIVE}..."
dd if=/tmp/fip-stm32mp257f-dk-optee-sdcard.bin of=/dev/${SD_DRIVE}p5 bs=1M conv=fsync
dd if=/tmp/fip-stm32mp257f-dk-optee-sdcard.bin of=/dev/${SD_DRIVE}p6 bs=1M conv=fsync
sync
reboot
